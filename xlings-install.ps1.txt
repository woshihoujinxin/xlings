#requires -version 5.1
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'

# 全局变量
$QI_RUN_DIR = (Get-Location).Path
$QI_INSTALL_DIR = '.xlings_software_install'
$SOFTWARE_URL0 = 'https://vip.123pan.cn/1825134841/32180721'
$SOFTWARE_URL1 = 'https://github.com/woshihoujinxin/xlings/archive/refs/heads/dev.zip'
$SOFTWARE_URL2 = 'https://gitee.com/houjinxin/xlings/repository/archive/dev.zip?format=zip'
$ZIP_FILE = 'software.zip'
$XLINGS_DIR = 'xlings-dev'
$INSTALL_SCRIPT_WIN = 'tools\install.win.bat'

# 简易彩色输出
function Write-Info([string]$msg)  { Write-Host $msg -ForegroundColor Cyan }
function Write-Ok([string]$msg)    { Write-Host $msg -ForegroundColor Green }
function Write-Warn([string]$msg)  { Write-Host $msg -ForegroundColor Yellow }
function Write-Err([string]$msg)   { Write-Host $msg -ForegroundColor Red }

# 错误退出
function Error-Exit([string]$msg) {
  Write-Err "错误: $msg"
  Cleanup
  throw $msg
}

# 清理临时目录
function Cleanup {
  try {
    if (Test-Path -LiteralPath (Join-Path $QI_RUN_DIR $QI_INSTALL_DIR)) {
      Write-Warn '清理临时文件...'
      Remove-Item -LiteralPath (Join-Path $QI_RUN_DIR $QI_INSTALL_DIR) -Recurse -Force -ErrorAction SilentlyContinue
    }
  } catch { }
}

# 测试命令是否存在
function Test-Command([string]$name) {
  try { $null = Get-Command $name -ErrorAction Stop; return $true } catch { return $false }
}

# 测量网络连通性（简化：能连即认为可用）
function Measure-Latency([string]$url) {
  try {
    $resp = Invoke-WebRequest -Uri $url -Method Head -TimeoutSec 10 -UseBasicParsing
    if ($resp.StatusCode -ge 200 -and $resp.StatusCode -lt 400) { return 0 } else { return 999999 }
  } catch { return 999999 }
}

# 源可用性检测：15s内响应且非网页（Content-Type != text/html）
function Test-Source([string]$url) {
  try {
    $resp = Invoke-WebRequest -Uri $url -Method Head -TimeoutSec 15 -UseBasicParsing
    if ($resp.StatusCode -ge 200 -and $resp.StatusCode -lt 400) {
      $ctype = $resp.Headers['Content-Type']
      if ($ctype -and ($ctype -like '*text/html*')) { return $false }
      return $true
    }
    return $false
  } catch { return $false }
}

# 按优先级选择源：VIP -> GitHub -> Gitee（先检查可用性）
function Select-FastestUrl {
  Write-Info '选择下载源（先检查可用性）...'

  if (Test-Source $SOFTWARE_URL0) {
    Write-Ok '使用 VIP 源'
    return $SOFTWARE_URL0
  } elseif (Test-Source $SOFTWARE_URL1) {
    Write-Ok '使用 GitHub 源'
    return $SOFTWARE_URL1
  } elseif (Test-Source $SOFTWARE_URL2) {
    Write-Ok '使用 Gitee 源'
    return $SOFTWARE_URL2
  } else {
    Write-Warn '源检测均失败，回退到 GitHub'
    return $SOFTWARE_URL1
  }
}

# 检查与安装必要工具（git）
function Ensure-Tool([string]$tool) {
  if (Test-Command $tool) {
    Write-Ok "$tool 已安装"
    return
  }
  Write-Warn "$tool 未安装，尝试安装..."
  if (Test-Command 'winget') {
    try {
      if ($tool -eq 'git') {
        winget install --id Git.Git --source winget --accept-source-agreements --accept-package-agreements | Out-Null
      }
    } catch { }
  } else {
    Write-Warn 'winget 不可用，请手动安装所需工具'
  }
  if (-not (Test-Command $tool)) { Error-Exit "安装 $tool 失败，请手动安装后重试" }
  Write-Ok "$tool 安装成功"
}

# 下载与安装
function Download-And-Install([string]$url) {
  # 清理旧安装目录
  if (Test-Path -LiteralPath $QI_INSTALL_DIR) {
    Write-Warn '清理旧的安装文件...'
    Remove-Item -LiteralPath $QI_INSTALL_DIR -Recurse -Force -ErrorAction SilentlyContinue
  }
  New-Item -ItemType Directory -Path $QI_INSTALL_DIR -Force | Out-Null
  Push-Location $QI_INSTALL_DIR

  # 下载（带回退）
  Write-Info '正在下载 xlings...'
  Write-Host ("下载地址: {0}" -f $url)
  Write-Host ("保存为:   {0}" -f $ZIP_FILE)

  $downloadOk = $false
  try {
    # 添加UserAgent避免被拦截
    $headers = @{
      'User-Agent' = 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
    }
    Invoke-WebRequest -Uri $url -OutFile $ZIP_FILE -Headers $headers -UseBasicParsing -TimeoutSec 120
    $downloadOk = $true
  } catch {
    Write-Warn "首选源下载失败: $($_.Exception.Message)"
    # 轮询候选源：VIP -> GitHub -> Gitee，跳过当前 URL；每个源在下载前做 15s 头部检查（非 text/html）
    $candidates = @($SOFTWARE_URL0, $SOFTWARE_URL1, $SOFTWARE_URL2)
    $switched = $false
    foreach ($candidate in $candidates) {
      if ($candidate -eq $url) { continue }
      Write-Host ("备用下载地址: {0}" -f $candidate)
      if (-not (Test-Source $candidate)) {
        Write-Warn "备用源不可用或返回网页，继续切换..."
        continue
      }
      try {
        $headers = @{
          'User-Agent' = 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
        }
        Invoke-WebRequest -Uri $candidate -OutFile $ZIP_FILE -Headers $headers -UseBasicParsing -TimeoutSec 120
        $downloadOk = $true
        $url = $candidate
        $switched = $true
        break
      } catch {
        Write-Warn "备用源下载失败: $($_.Exception.Message)"
      }
    }
    if (-not $switched) {
      Error-Exit "下载失败，所有源均不可用，请检查网络后重试"
    }
  }

  if (-not (Test-Path -LiteralPath $ZIP_FILE)) { Error-Exit '下载的文件不存在' }
  if ((Get-Item $ZIP_FILE).Length -le 0) { Error-Exit '下载的文件为空' }

  # 验证是否为有效的ZIP文件
  $fileHeader = Get-Content $ZIP_FILE -Encoding Byte -TotalCount 4
  if ($fileHeader[0] -ne 0x50 -or $fileHeader[1] -ne 0x4B) {
    Write-Err "下载的文件不是有效的ZIP格式，文件大小: $((Get-Item $ZIP_FILE).Length) 字节"
    Write-Host "文件开头内容:"
    Get-Content $ZIP_FILE -TotalCount 5
    Error-Exit '下载的文件格式无效，可能是网络返回的错误页面'
  }

  # 解压
  Write-Info '正在解压文件...'
  Write-Host ("解压源:   {0}" -f $ZIP_FILE)
  Write-Host ("目标目录: {0}" -f $QI_INSTALL_DIR)
  try {
    Expand-Archive -LiteralPath $ZIP_FILE -DestinationPath . -Force
  } catch { Error-Exit '解压失败，文件可能已损坏' }

  # 进入解压目录
  Write-Host ("进入目录: {0}" -f $XLINGS_DIR)
  if (-not (Test-Path -LiteralPath $XLINGS_DIR)) { Error-Exit "找不到解压后的目录 $XLINGS_DIR" }
  Push-Location $XLINGS_DIR

  # 检查安装脚本（Windows）
  if (-not (Test-Path -LiteralPath $INSTALL_SCRIPT_WIN)) { Error-Exit "找不到安装脚本 $INSTALL_SCRIPT_WIN" }

  # 显式导出环境变量，供子脚本使用
  try {
    if (-not $env:XLINGS_HOME_DIR) {
      $env:XLINGS_HOME_DIR = $env:USERPROFILE
    }
  } catch {
    Write-Warn "无法设置环境变量: $_"
  }

  # 运行安装脚本（cmd 执行 .bat）
  Write-Info '正在运行安装脚本...'
  Write-Host ("执行脚本: {0}" -f $INSTALL_SCRIPT_WIN)
  $batPath = (Resolve-Path $INSTALL_SCRIPT_WIN).Path
  $proc = Start-Process -FilePath 'cmd.exe' -ArgumentList "/c `"$batPath`" disable_reopen" -NoNewWindow -Wait -PassThru
  if ($proc.ExitCode -ne 0) { Error-Exit '安装脚本执行失败' }

  # 安装后校验
  if (-not (Test-Command 'xlings')) {
    Write-Warn 'xlings 未在 PATH 中，尝试追加路径并重试...'
    $env:PATH = "$($env:USERPROFILE)\.xlings_data\bin;$env:PATH"
    if (-not (Test-Command 'xlings')) {
      Write-Err 'xlings 仍不可用。请重开终端或手动将以下路径加入 PATH：'
      Write-Host "$($env:USERPROFILE)\.xlings_data\bin"
    }
  }
}

# 配置环境变量（持久）
function Setup-Environment {
  Write-Info '配置环境变量...'
  $userHome = $env:USERPROFILE
  try {
    if ([string]::IsNullOrWhiteSpace($env:XLINGS_HOME_DIR)) { $env:XLINGS_HOME_DIR = $userHome }
  } catch {
    Write-Warn "环境变量设置警告: $_"
  }

  # 更新用户 PATH（避免重复）
  $bin1 = Join-Path $userHome '.xlings\bin'
  $bin2 = Join-Path $userHome '.xlings_data\bin'
  $userPath = [Environment]::GetEnvironmentVariable('Path', 'User')
  if (-not ($userPath -like "*$bin1*")) { $userPath = "$bin1;$userPath" }
  if (-not ($userPath -like "*$bin2*")) { $userPath = "$bin2;$userPath" }
  [Environment]::SetEnvironmentVariable('Path', $userPath, 'User')
  Write-Ok '环境变量已配置'

  # 当前会话立即生效
  $env:PATH = "$bin1;$bin2;$($env:PATH)"
}

# 主流程
function Main {
  Show-Banner
  Write-Info '开始安装 xlings...'

  # 依赖（git）
  Ensure-Tool 'git'

  # 选择源
  $softwareUrl = Select-FastestUrl

  # 下载与安装
  Download-And-Install -url $softwareUrl

  # 配置环境
  Setup-Environment

  # 清理
  Cleanup

  Write-Ok '安装完成！'
  Write-Warn '请重新启动终端使环境变量生效'
  Write-Host ''
  Write-Info "运行 'xlings help' 获取更多信息"
}

# 运行
Main
